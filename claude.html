import anthropic

import os
import time

client = anthropic.Anthropic()




instructions = "Which intent does this bank customer's message express?"

label_list = "\n".join(f"- {k}: {v}" for k, v in labels_dict.items())






tool = {
    "name": "classify",
    "description": "Record the intent label for the message.",
    "input_schema": {
        "type": "object",
        "properties": {
            "label": {"type": "string", "enum": list(labels_dict.keys())},
            "confidence": {
                "type": "number",
                "description": "Probability from 0 to 1 that the label is correct."
            }
        },
        "required": ["label", "confidence"]
    }
}


if os.path.exists("results/claude.jsonl"):

    done_ids = set(pd.read_json("results/claude.jsonl", lines=True)["id"])
else:
    done_ids = set()

with open("results/claude.jsonl", "a", encoding="utf-8") as file:

    for row in test.itertuples():

        if row.id in done_ids:
            continue
    

        id = row.id
        text = row.text
        true_label = row.label


        prompt = f"""{instructions}

        Choose exactly one label from this list:
        {label_list}

        Message:
        <message>
        {text}
        </message>

        Also give your confidence: the probability from 0 to 1 that your chosen label is correct."""

        

        start_time = time.perf_counter()


        message = client.messages.create(
            model="claude-haiku-5-5",  
            max_tokens=1024,
            messages=[
                {"role": "user", "content": prompt}
            ],
            tools=[tool],
            tool_choice={"type": "tool", "name": "classify"}
        )

        end_time = time.perf_counter()
        run_time = end_time - start_time


        prediction = message.content[0].input["label"]
        confidence = message.content[0].input["confidence"]
        input_tokens = message.usage.input_tokens
        output_tokens = message.usage.output_tokens
        model = message.model
        

        json_dict = {
                "id": id,
                "label": true_label,
                "pred": prediction,
                "prob": confidence,
                "latency": run_time,
                "input_tokens": input_tokens,
                "output_tokens":output_tokens,
                "model": model
            }

        j_string = json.dumps(json_dict)
        file.write(j_string + "\n")
